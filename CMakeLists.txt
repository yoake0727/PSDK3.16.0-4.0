cmake_minimum_required(VERSION 3.15)
project(psdk_payload C CXX)
if (CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "不允许源码内构建，请使用独立的 build 目录。")
endif ()
set(USE_SYSTEM_ARCH "LINUX" CACHE STRING "目标平台（LINUX 或 RTOS）")
if (NOT USE_SYSTEM_ARCH STREQUAL "LINUX")
    message(FATAL_ERROR "当前目录结构仅支持 USE_SYSTEM_ARCH=LINUX。")
endif ()
add_compile_definitions(SYSTEM_ARCH_LINUX)
add_subdirectory(src)
