cmake_minimum_required(VERSION 3.15)
project(psdk_payload_app C CXX)

set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_compile_definitions(_GNU_SOURCE PLATFORM_ARCH_AARCH64=1 SYSTEM_ARCH_LINUX=1)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(PSDK_INCLUDE "${ROOT}/psdk_lib/include")
set(PSDK_LIB "${ROOT}/psdk_lib/lib/aarch64-linux-gnu-gcc/libpayloadsdk.a")
set(DEPS "${ROOT}/third_party/aarch64")
set(MOSQ "${DEPS}/lib/libmosquitto.a")
set(FFMPEG "${DEPS}/lib/libavformat.a" "${DEPS}/lib/libavcodec.a" "${DEPS}/lib/libswscale.a" "${DEPS}/lib/libavutil.a")
foreach (f "${PSDK_INCLUDE}/dji_core.h" "${PSDK_LIB}" "${DEPS}/include/mosquitto.h" "${MOSQ}" ${FFMPEG})
    if (NOT EXISTS "${f}")
        message(FATAL_ERROR "缺少构建依赖：${f}")
    endif ()
endforeach ()
file(GLOB_RECURSE APP CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/application/*.c" "${CMAKE_CURRENT_LIST_DIR}/application/*.cpp")
file(GLOB_RECURSE MOD CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/module_sample/*.c" "${CMAKE_CURRENT_LIST_DIR}/module_sample/*.cpp")
file(GLOB_RECURSE COMMON CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/common/*.c" "${CMAKE_CURRENT_LIST_DIR}/common/*.cpp")
file(GLOB_RECURSE HAL CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/hal/*.c" "${CMAKE_CURRENT_LIST_DIR}/hal/*.cpp")
add_executable(${PROJECT_NAME} ${APP} ${MOD} ${COMMON} ${HAL})
target_include_directories(${PROJECT_NAME} PRIVATE "${PSDK_INCLUDE}" "${DEPS}/include" "${CMAKE_CURRENT_LIST_DIR}/application" "${CMAKE_CURRENT_LIST_DIR}/common" "${CMAKE_CURRENT_LIST_DIR}/module_sample")
find_package(OpenCV QUIET)
if (OpenCV_FOUND)
    target_compile_definitions(${PROJECT_NAME} PRIVATE OPEN_CV_INSTALLED)
    target_include_directories(${PROJECT_NAME} PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(${PROJECT_NAME} PRIVATE ${OpenCV_LIBS})
endif ()
target_link_libraries(${PROJECT_NAME} PRIVATE "-Wl,--start-group" "${PSDK_LIB}" ${FFMPEG} "${MOSQ}" "-Wl,--end-group" pthread ssl crypto m dl stdc++)
set_target_properties(${PROJECT_NAME} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

set(APP_JSON "${CMAKE_CURRENT_LIST_DIR}/app_json/app.json")
set(BUILD_DPK "${ROOT}/tools/build_dpk/build_dpk.sh")
set(DPK_OUTPUT "${CMAKE_BINARY_DIR}/dpk")
if (NOT EXISTS "${APP_JSON}" OR NOT EXISTS "${BUILD_DPK}")
    message(FATAL_ERROR "缺少打包所需文件：app.json 或 build_dpk.sh")
endif ()
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${DPK_OUTPUT}"
    COMMAND bash "${BUILD_DPK}" -i "${APP_JSON}" -o "${DPK_OUTPUT}"
    DEPENDS "${APP_JSON}" "${BUILD_DPK}"
    COMMENT "正在生成 DJI DPK 软件包")
